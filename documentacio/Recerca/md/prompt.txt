    def _meshes_intersect_fast(self, mesh1: trimesh.Trimesh, mesh2: trimesh.Trimesh) -> bool:
        """Verificació ràpida si dues malles es toquen"""
        # Primer: verificar distància entre centres
        center1 = mesh1.vertices.mean(axis=0)
        center2 = mesh2.vertices.mean(axis=0)
        distance = np.linalg.norm(center1 - center2)
        
        # Estimar radi màxim de cada malla
        radius1 = np.linalg.norm(mesh1.vertices - center1, axis=1).max()
        radius2 = np.linalg.norm(mesh2.vertices - center2, axis=1).max()

        if distance > (radius1 + radius2 + self.collision_tolerance):
            return False
        
        # Verificació més precisa: vèrtexs dins de l'altra malla
        return self._vertices_inside_mesh(mesh1.vertices, mesh2) or \
               self._vertices_inside_mesh(mesh2.vertices, mesh1)

            print(f"❌ Error en packing real: {e}")
            return {
                'success': False,
                'pieces_count': 0,
                'positions': [],
                'orientations': [],
                'meshes': [],
                'efficiency': 0,
                'execution_time': time.time() - start_time,
                'method': 'Real Mesh Packing',
                'error': str(e)
            }
    
    def _simplify_mesh_for_collision(self, mesh: trimesh.Trimesh) -> trimesh.Trimesh:
        """Simplifica la malla per optimitzar detecció de col·lisions"""
        try:
            # Si ja té poques cares, no cal simplificar
            if len(mesh.faces) <= MESH_REDUCTION_TARGET:
                print(f"   ℹ️ Malla ja té {len(mesh.faces)} cares, no cal simplificar")
                return mesh.copy()
            
            # Usar el mateix sistema que packassist
            simplified = mesh.simplify_quadric_decimation(MESH_REDUCTION_TARGET)
            
            # Verificar que la simplificació ha funcionat
            if simplified is None or len(simplified.faces) == 0:
                print(f"   ⚠️ Simplificació ha fallat, usant malla original")
                return mesh.copy()
            
            print(f"   ✅ Malla simplificada: {len(mesh.faces)} → {len(simplified.faces)} cares")
            return simplified
            
        except Exception as e:
            print(f"   ⚠️ Error simplificant malla: {e}, usant original")
            return mesh.copy()
    
    def _generate_real_orientations(self, count: int) -> List[np.ndarray]:
        """Genera orientacions REALS, no només 90° graus"""
        orientations = []
        
        # Orientacions bàsiques (per començar)
        basic_angles = [
            [0, 0, 0], [90, 0, 0], [0, 90, 0], [0, 0, 90],
            [180, 0, 0], [0, 180, 0], [0, 0, 180],
            [90, 90, 0], [90, 0, 90], [0, 90, 90],
            [45, 0, 0], [0, 45, 0], [0, 0, 45],
            [30, 60, 0], [60, 30, 0], [45, 45, 45]
        ]
        
        for angles in basic_angles:
            matrix = trimesh.transformations.compose_matrix(
                angles=[np.radians(a) for a in angles]
            )
            orientations.append(matrix)
        
        # Orientacions aleatòries per diversitat
        np.random.seed(42)  # Per reproducibilitat
        remaining = count - len(orientations)
        
        for _ in range(max(0, remaining)):
            # Angles aleatoris més naturals
            angles = [
                np.random.uniform(0, 360),  # X: rotació completa
                np.random.uniform(0, 360),  # Y: rotació completa  
                np.random.uniform(0, 360)   # Z: rotació completa
            ]
            
            matrix = trimesh.transformations.compose_matrix(
                angles=[np.radians(a) for a in angles]
            )
            orientations.append(matrix)
        
        return orientations[:count]
    
    def _pack_oriented_mesh(self, oriented_mesh: trimesh.Trimesh, target_pieces: int) -> Dict[str, Any]:
        """Col·loca peces d'una orientació específica"""
        positions = []
        orientations = []
        meshes = []
        placed_mesh_data = []  # (mesh, transform)
        
        # Calcular la caixa envolvente només per saber límits de cerca
        bounds = oriented_mesh.bounds
        search_dims = bounds[1] - bounds[0]
        
        # Estratègia: provar posicions des del terra cap amunt
        max_layers = int(self.container_dims[2] / search_dims[2]) + 1
        
        for layer in range(max_layers):
            layer_z = search_dims[2]/2 + layer * search_dims[2]
            
            # Si la capa no cap, parem
            if layer_z + search_dims[2]/2 > self.container_dims[2]:
                break
            
            # Grid de posicions en aquesta capa
            step_x = max(5, search_dims[0] / 3)  # Passos més petits per millor packing
            step_y = max(5, search_dims[1] / 3)
            
            x_positions = np.arange(search_dims[0]/2, self.container_dims[0] - search_dims[0]/2, step_x)
            y_positions = np.arange(search_dims[1]/2, self.container_dims[1] - search_dims[1]/2, step_y)
            
            for x in x_positions:
                for y in y_positions:
                    if len(positions) >= target_pieces:
                        break
                    
                    # Crear transform per aquesta posició
                    translation = np.eye(4)
                    translation[:3, 3] = [x, y, layer_z]
                    
                    # Aplicar transform a la malla
                    test_mesh = oriented_mesh.copy()
                    test_mesh.apply_transform(translation)
                    
                    # Verificar que cap dins del contenidor
                    if not self._mesh_fits_in_container(test_mesh):
                        continue
                    
                    # Verificar col·lisions amb peces ja col·locades
                    if self._check_mesh_collisions(test_mesh, placed_mesh_data):
                        continue
                    
                    # Si arriba aquí, la posició és vàlida
                    positions.append([x, y, layer_z])
                    orientations.append(np.eye(4))  # Orientació ja aplicada a oriented_mesh
                    meshes.append(test_mesh.copy())
                    placed_mesh_data.append(test_mesh)
                    
                    print(f"      ✅ Peca {len(positions)} col·locada a ({x:.1f}, {y:.1f}, {layer_z:.1f})")
        
        return {
            'positions': positions,
            'orientations': orientations,
            'meshes': meshes,
            'count': len(positions)
        }
    
    def _mesh_fits_in_container(self, mesh: trimesh.Trimesh) -> bool:
        """Verifica que la malla REAL cap dins del contenidor"""
        bounds = mesh.bounds
        
        return (bounds[0][0] >= 0 and bounds[1][0] <= self.container_dims[0] and
                bounds[0][1] >= 0 and bounds[1][1] <= self.container_dims[1] and
                bounds[0][2] >= 0 and bounds[1][2] <= self.container_dims[2])
    
    def _check_mesh_collisions(self, test_mesh: trimesh.Trimesh, placed_meshes: List[trimesh.Trimesh]) -> bool:
        """Detecció de col·lisions REAL entre malles"""
        for placed_mesh in placed_meshes:
            try:
                # Usar la funció de col·lisions de trimesh
                collision_manager = trimesh.collision.CollisionManager()
                collision_manager.add_object('placed', placed_mesh)
                
                # Verificar col·lisió
                is_collision = collision_manager.in_collision_single(test_mesh)
                
                if is_collision:
                    return True
                    
                # També verificar distància mínima
                distance = self._mesh_distance(test_mesh, placed_mesh)
                if distance < self.collision_tolerance:
                    return True
                    
            except Exception as e:
                # Si falla la detecció avançada, usar bounds com a fallback
                return self._bounds_collision(test_mesh, placed_mesh)
        
        return False
    
    def _mesh_distance(self, mesh1: trimesh.Trimesh, mesh2: trimesh.Trimesh) -> float:
        """Calcula distància mínima entre dues malles"""
        try:
            # Usar centres de massa com aproximació
            center1 = mesh1.center_mass
            center2 = mesh2.center_mass
            return np.linalg.norm(center1 - center2)
        except:
            # Fallback: distància entre centres de bounds
            return np.linalg.norm(mesh1.bounds.mean(axis=0) - mesh2.bounds.mean(axis=0))
    
    def _bounds_collision(self, mesh1: trimesh.Trimesh, mesh2: trimesh.Trimesh) -> bool:
        """Detecció de col·lisions per bounds (fallback)"""
        bounds1 = mesh1.bounds
        bounds2 = mesh2.bounds
        
        return not (bounds1[1][0] <= bounds2[0][0] or bounds1[0][0] >= bounds2[1][0] or
                   bounds1[1][1] <= bounds2[0][1] or bounds1[0][1] >= bounds2[1][1] or
                   bounds1[1][2] <= bounds2[0][2] or bounds1[0][2] >= bounds2[1][2])

    def start_real_packing(self):

        if not self.mesh:

            return
        
        self.start_btn.config(state='disabled')
        self.progress.start()
        self.progress_label.config(text="Executant packing REAL...")
        
        def packing_worker():
            try:
                packer = RealMeshPacker(CONTAINER_DIMS)
                result = packer.pack_real_meshes(self.mesh, auto_max_pieces=True)  # Mode automàtic
